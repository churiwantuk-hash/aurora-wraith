# Removes the pasted iwr|iex command from PSReadLine's saved history and the session's up-arrow history.
function Clear-CommandTrace {
    if (($Sandbox -or $AutoAction) -and -not $env:NEXUS_HISTORY_TEST) { return }
    $marker = '(?i)(AuroraRunAll|NexusRunAll|AuroraWraith|NEXUS-FiveM|aurora|wraith)'
    $fetch = '(?i)(\biwr\b|\birm\b|invoke-webrequest|invoke-restmethod|\biex\b|invoke-expression)'
    try {
        $path = (Get-PSReadLineOption -ErrorAction Stop).HistorySavePath
        if ($path -and (Test-Path $path)) {
            $lines = [IO.File]::ReadAllLines($path)
            $keep = @($lines | Where-Object { -not ($_ -match $fetch -and $_ -match $marker) })
            if ($keep.Count -ne $lines.Count) { [IO.File]::WriteAllLines($path, [string[]]$keep) }
        }
    } catch { }
    try { [Microsoft.PowerShell.PSConsoleReadLine]::ClearHistory() } catch { }
}

# ------------------------------------------------------------------ hardware detection
function Get-Hardware {
    $cpu = (Get-CimInstance Win32_Processor | Select-Object -First 1)
    $os = Get-CimInstance Win32_OperatingSystem
    $ramGb = [math]::Round((Get-CimInstance Win32_ComputerSystem).TotalPhysicalMemory / 1GB, 1)

    # VRAM from the display class registry key (Win32_VideoController caps at 4 GB)
    $vramByName = @{}
    $cls = 'HKLM:\SYSTEM\CurrentControlSet\Control\Class\{4d36e968-e325-11ce-bfc1-08002be10318}'
    Get-ChildItem $cls -ErrorAction SilentlyContinue | Where-Object { $_.PSChildName -match '^\d{4}$' } | ForEach-Object {
        try {
            $k = Get-ItemProperty $_.PSPath -ErrorAction Stop
            $q = $k.'HardwareInformation.qwMemorySize'
            if (-not $q) {
                # Older drivers store a DWORD or 4-byte binary instead of the 64-bit value
                $m = $k.'HardwareInformation.MemorySize'
                if ($m -is [byte[]] -and $m.Length -ge 4) { $q = [BitConverter]::ToUInt32($m, 0) } elseif ($m) { $q = [uint32]$m }
            }
            if ($k.DriverDesc -and $q) { $vramByName[$k.DriverDesc] = [math]::Round([double]$q / 1GB, 1) }
        } catch { }
    }
    $gpus = @()
    foreach ($v in (Get-CimInstance Win32_VideoController)) {
        $pci = [string]$v.PNPDeviceID -like 'PCI\*'
        $vendor = 'Other'
        if ($v.PNPDeviceID -match 'VEN_10DE') { $vendor = 'NVIDIA' } elseif ($v.PNPDeviceID -match 'VEN_1002') { $vendor = 'AMD' } elseif ($v.PNPDeviceID -match 'VEN_8086') { $vendor = 'Intel' }
        $vram = 0
        if ($vramByName.ContainsKey($v.Name)) { $vram = $vramByName[$v.Name] } elseif ($v.AdapterRAM) { $vram = [math]::Round([double]$v.AdapterRAM / 1GB, 1) }
        # Integrated GPUs: Intel UHD/Iris (not Arc), AMD APU graphics (Radeon Graphics, Vega 8, 680M/780M/890M...)
        $amdIgpu = $v.Name -match 'Radeon\(TM\) Graphics|Radeon Graphics|Vega \d+|Radeon\(TM\) \d{3}M|Radeon \d{3}M'
        $integrated = $pci -and (($vendor -eq 'Intel' -and $v.Name -notmatch 'Arc') -or ($vendor -eq 'AMD' -and ($amdIgpu -or ($vram -lt 2 -and $v.Name -notmatch '\bRX\b'))) -or ($vram -gt 0 -and $vram -lt 1))
        $gpus += [pscustomobject]@{ Name = $v.Name; Vendor = $vendor; VramGb = $vram; IsPci = $pci; Integrated = $integrated; Driver = $v.DriverVersion }
    }
    $primary = $gpus | Where-Object { $_.IsPci -and -not $_.Integrated } | Sort-Object VramGb -Descending | Select-Object -First 1
    if (-not $primary) { $primary = $gpus | Where-Object { $_.IsPci } | Select-Object -First 1 }
    $hybrid = (@($gpus | Where-Object { $_.IsPci }).Count -ge 2) -and [bool]($gpus | Where-Object { $_.IsPci -and -not $_.Integrated })

    $battery = @(Get-CimInstance Win32_Battery -ErrorAction SilentlyContinue).Count -gt 0
    $onAc = $true
    if ($battery) { try { $onAc = [bool](Get-CimInstance -Namespace root\wmi -ClassName BatteryStatus -ErrorAction Stop | Select-Object -First 1).PowerOnline } catch { } }

    $cur = [NexusNative]::Current()
    $w = 0; $h = 0; $hz = 0; $maxHz = 0
    if ($cur) { $w = $cur[0]; $h = $cur[1]; $hz = $cur[2]; $maxHz = [math]::Max($hz, [NexusNative]::MaxRefresh($w, $h)) }

    $vramPrimary = 0
    if ($primary -and -not $primary.Integrated) { $vramPrimary = $primary.VramGb }
    $threads = [Environment]::ProcessorCount
    $tier = 'MID'
    if ($vramPrimary -lt 3.5 -or $ramGb -lt 7.5 -or $threads -lt 4) { $tier = 'LOW' }
    elseif ($vramPrimary -ge 7.5 -and $ramGb -ge 15 -and $threads -ge 8) { $tier = 'HIGH' }

    $ssd = $null
    try {
        $letter = $env:LOCALAPPDATA.Substring(0, 1)
        $part = Get-Partition -DriveLetter $letter -ErrorAction Stop
        $disk = Get-PhysicalDisk -ErrorAction Stop | Where-Object { $_.DeviceId -eq [string]$part.DiskNumber }
        if ($disk) { $ssd = ($disk.MediaType -eq 'SSD') -or ($disk.BusType -eq 'NVMe') }
    } catch { }

    return [pscustomobject]@{
        Cpu = ($cpu.Name -replace '\s+', ' ').Trim(); Threads = $threads; RamGb = $ramGb
        Gpus = $gpus; Primary = $primary; VramGb = $vramPrimary; Hybrid = $hybrid
        Laptop = $battery; OnAc = $onAc
        Width = $w; Height = $h; Hz = $hz; MaxHz = $maxHz
        Windows = "$($os.Caption) (build $($os.BuildNumber))"; Build = [int]$os.BuildNumber
        Tier = $tier; Ssd = $ssd
    }
}

# ------------------------------------------------------------------ value helpers
function Same($a, $b) {
    if ($null -eq $a -or $null -eq $b) { return ($null -eq $a -and $null -eq $b) }
    $x = 0.0; $y = 0.0
    if ([double]::TryParse([string]$a, [Globalization.NumberStyles]::Float, $Inv, [ref]$x) -and [double]::TryParse([string]$b, [Globalization.NumberStyles]::Float, $Inv, [ref]$y)) {
        return ([math]::Abs($x - $y) -lt 0.0001)
    }
    return ([string]$a).Trim() -ieq ([string]$b).Trim()
}
function F([double]$v) { return $v.ToString('0.000000', $Inv) }

function Get-GtaPreset([string]$mode, $hw) {
    $v = $hw.VramGb
    $texSafe = '0'; if ($v -ge 10) { $texSafe = '2' } elseif ($v -ge 5.5) { $texSafe = '1' }
    $texBal = '0'; if ($v -ge 7.5) { $texBal = '2' } elseif ($v -ge 3.5) { $texBal = '1' }
    $af = '4'; if ($hw.Tier -eq 'LOW') { $af = '2' }
    $d = [ordered]@{
        'configSource' = 'SMC_USER'; 'DX_Version' = '2'; 'TextureQuality' = $texSafe; 'ShaderQuality' = '0'; 'ShadowQuality' = '1'
        'ReflectionQuality' = '0'; 'ReflectionMSAA' = '0'; 'WaterQuality' = '0'; 'ParticleQuality' = '0'; 'GrassQuality' = '0'
        'SSAO' = '0'; 'AnisotropicFiltering' = $af; 'MSAA' = '0'; 'Tessellation' = '0'; 'PostFX' = '0'; 'Shadow_SoftShadows' = '0'
        'FXAA_Enabled' = 'true'; 'TXAA_Enabled' = 'false'; 'UltraShadows_Enabled' = 'false'; 'Shadow_LongShadows' = 'false'
        'Shadow_ParticleShadows' = 'false'; 'DoF' = 'false'; 'HdStreamingInFlight' = 'false'; 'MotionBlurStrength' = (F 0)
        'LodScale' = (F 0); 'MaxLodScale' = (F 0); 'PedLodBias' = (F 0); 'VehicleLodBias' = (F 0); 'CityDensity' = (F 0)
        'PedVarietyMultiplier' = (F 0); 'VehicleVarietyMultiplier' = (F 0); 'Windowed' = '0'; 'VSync' = '0'
    }
    if ($hw.MaxHz -gt 0) { $d['RefreshRate'] = [string]$hw.MaxHz }
    switch ($mode) {
        'MAXFPS' { if ($v -ge 5.5) { $d['TextureQuality'] = '1' } else { $d['TextureQuality'] = '0' } }
        'SMOOTH' { $d['LodScale'] = (F 0.2) }
        'COMPETITIVE' { $d['LodScale'] = (F 0.3); $d['FXAA_Enabled'] = 'false' }
        'EXTREME' {
            # Crowd settings with Normal textures regardless of VRAM.
            $d['TextureQuality'] = '0'; $d['FXAA_Enabled'] = 'false'
        }
        'OMEGA' {
            # Absolute minimum on every setting: the lightest load the game can run with.
            $d['TextureQuality'] = '0'; $d['AnisotropicFiltering'] = '0'; $d['FXAA_Enabled'] = 'false'
        }
        'CROWD' {
            # Crowded servers are CPU/streaming-bound: minimum draw distance + population, and textures low enough
            # that dozens of custom cars/clothes fit in VRAM without streaming stalls. FXAA off for a sharp image.
            if ($v -ge 10) { $d['TextureQuality'] = '1' } else { $d['TextureQuality'] = '0' }
            $d['FXAA_Enabled'] = 'false'
        }
        'BALANCED' {
            $s = ($hw.Tier -eq 'HIGH')
            $d['TextureQuality'] = $texBal
            foreach ($k in 'ShaderQuality', 'ReflectionQuality', 'GrassQuality', 'SSAO', 'Tessellation', 'PostFX', 'Shadow_SoftShadows') { if ($s) { $d[$k] = '1' } }
            if ($s) { $d['ShadowQuality'] = '2' }
            $d['WaterQuality'] = '1'; $d['ParticleQuality'] = '1'; $d['Shadow_ParticleShadows'] = 'true'
            $d['LodScale'] = (F 0.6); $d['PedLodBias'] = (F 0.2); $d['VehicleLodBias'] = (F 0.2)
            $d['CityDensity'] = (F 0.5); $d['PedVarietyMultiplier'] = (F 0.5); $d['VehicleVarietyMultiplier'] = (F 0.5)
            $d.Remove('Windowed')
        }
    }
    return $d
}

function Get-Plan([string]$profile, [string]$mode, $hw) {
    $plan = @()
    $plan += New-Reg 'gamemode' 'Game Mode' 'HKCU' 'Software\Microsoft\GameBar' 'AutoGameModeEnabled' 'DWord' '1' '1' 'Windows prioritises the game and holds back Windows Update installs while playing'
    $plan += New-Reg 'gamedvr' 'Game DVR (background recording)' 'HKCU' 'System\GameConfigStore' 'GameDVR_Enabled' 'DWord' '0' '1' 'Stops background game recording that costs CPU, GPU and RAM'
    $plan += New-Reg 'appcapture' 'App capture (Game Bar)' 'HKCU' 'Software\Microsoft\Windows\CurrentVersion\GameDVR' 'AppCaptureEnabled' 'DWord' '0' '1' 'Stops Game Bar background capture'
    # Windows defaults that tweak packs often disable. Without them a 16 GB PC pages to disk instead of compressing in RAM.
    $plan += [pscustomobject]@{ Id = 'sysmain'; Title = 'SysMain service'; Kind = 'svc'; Name = 'SysMain'; Proposed = 'Automatic'; Default = $null
        Why = 'Required by Memory Compression (Windows default: enabled)'; Admin = $true }
    $plan += [pscustomobject]@{ Id = 'memcomp'; Title = 'Memory Compression'; Kind = 'mmagent'; Proposed = 'true'; Default = $null
        Why = 'Keeps data compressed in RAM instead of paging to disk, which causes stutter when RAM is full (Windows default)'; Admin = $true }
    if ($hw.Hybrid) {
        foreach ($exe in (Get-FiveMExes)) {
            $plan += New-Reg ('gpupref:' + $exe) ('Dedicated GPU for ' + (Split-Path $exe -Leaf)) 'HKCU' 'Software\Microsoft\DirectX\UserGpuPreferences' $exe 'String' 'GpuPreference=2;' $null 'This PC also has an integrated GPU, so FiveM is pinned to the dedicated GPU'
        }
    }
    if ($hw.MaxHz -gt $hw.Hz) {
        $plan += [pscustomobject]@{ Id = 'refresh'; Title = 'Display refresh rate'; Kind = 'display'; Proposed = [string]$hw.MaxHz; Default = $null; Why = "Monitor supports $($hw.MaxHz) Hz but runs at $($hw.Hz) Hz"; Admin = $false }
    }
    if ($profile -ne 'LOW' -or $mode -eq 'COMPETITIVE') {
        $plan += [pscustomobject]@{ Id = 'mouse'; Title = 'Mouse acceleration'; Kind = 'mouse'; Proposed = '0,0,0'; Default = $null; Why = '1:1 mouse movement for consistent aim'; Admin = $false }
    }
    if ($profile -eq 'HIGH' -or $profile -eq 'MAX') {
        $ult = ($profile -eq 'MAX' -and -not $hw.Laptop)
        $pwWhy = 'Stops the CPU from downclocking between frames (only replaces stock Windows plans)'
        if ($ult) { $pwWhy = 'Ultimate Performance: CPU never downclocks, lower frame latency (desktops only)' }
        $plan += [pscustomobject]@{ Id = 'power'; Title = 'Power plan'; Kind = 'power'; Proposed = $null; Default = $null; Why = $pwWhy; Admin = $false; Ultimate = $ult }
    }
    if ($profile -eq 'MAX' -and $hw.Build -ge 19041 -and $hw.Primary -and $hw.Primary.Vendor -in @('NVIDIA', 'AMD')) {
        $plan += New-Reg 'hags' 'HAGS (GPU scheduling)' 'HKLM' 'SYSTEM\CurrentControlSet\Control\GraphicsDrivers' 'HwSchMode' 'DWord' '2' '1' 'Can lower latency on some PCs, no difference on others (after reboot)' -Admin
    }
    if ($profile -eq 'MAX') {
        foreach ($exe in @(Get-FiveMExes | Where-Object { $_ -like '*GTAProcess.exe' })) {
            # Value is computed in Build-Preview so existing compatibility flags on the exe are kept.
            $plan += New-Reg ('fso:' + $exe) ('Fullscreen Optimizations off: ' + (Split-Path $exe -Leaf)) 'HKCU' 'Software\Microsoft\Windows NT\CurrentVersion\AppCompatFlags\Layers' $exe 'String' 'FSO' $null 'True exclusive fullscreen, can lower input lag on some PCs'
        }
    }
    if ($mode -ne 'NONE') {
        $preset = Get-GtaPreset $mode $hw
        foreach ($k in $preset.Keys) {
            $section = 'graphics'
            if ($k -eq 'configSource') { $section = '' } elseif ($VideoElements -contains $k) { $section = 'video' }
            $plan += [pscustomobject]@{ Id = 'gta.' + $k; Title = 'GTA: ' + $k; Kind = 'xml'; Element = $k; Section = $section; Proposed = $preset[$k]; Default = $null; Why = ''; Admin = $false }
        }
    }
    return $plan
}

# ------------------------------------------------------------------ read / write per kind
function Resolve-RegTarget($s) {
    if ($Sandbox) { return @{ Root = [Microsoft.Win32.Registry]::CurrentUser; Sub = 'Software\NexusFiveM.Tests\' + $s.Hive + '\' + $s.SubKey } }
    if ($s.Hive -eq 'HKLM') { return @{ Root = [Microsoft.Win32.Registry]::LocalMachine; Sub = $s.SubKey } }
    return @{ Root = [Microsoft.Win32.Registry]::CurrentUser; Sub = $s.SubKey }
}

function Get-PowerActive {
    $out = (powercfg /getactivescheme) | Out-String
    if ($out -match '([0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12})') { return $Matches[1].ToLower() }
    return $null
}
$PlanBalanced = '381b4222-f694-41f0-9685-ff5bb260df2e'
$PlanSaver = 'a1841308-3541-4fab-bc81-f71556f20b4a'
$PlanHigh = '8c5e7fda-e8bf-4a96-9a85-a6e23a8c635c'

function Read-Value($s) {
    switch ($s.Kind) {
        'reg' {
            $t = Resolve-RegTarget $s
            $k = $t.Root.OpenSubKey($t.Sub)
            if (-not $k) { return $null }
            try { $v = $k.GetValue($s.Name, $null) } finally { $k.Close() }
            if ($null -eq $v) { return $null }
            return [string]$v
        }
        'xml' {
            if (-not (Test-Path $GtaXml)) { return $null }
            $doc = New-Object Xml.XmlDocument
            $doc.Load($GtaXml)
            $node = $null
            if ($s.Section -eq '') { $node = $doc.DocumentElement.SelectSingleNode($s.Element) } else { $node = $doc.DocumentElement.SelectSingleNode($s.Section + '/' + $s.Element) }
            if (-not $node) { return $null }
            if ($s.Element -eq 'configSource') { return $node.InnerText }
            return $node.GetAttribute('value')
        }
        'mouse' { if ($Sandbox) { return $script:SandboxMouse } return [NexusNative]::GetMouse() }
        'display' { if ($Sandbox) { return $script:SandboxHz } $c = [NexusNative]::Current(); if ($c) { return [string]$c[2] } return $null }
        'power' { if ($Sandbox) { return $script:SandboxPower } return Get-PowerActive }
        'svc' {
            if ($Sandbox) { return $script:SandboxSvc }
            $svc = Get-Service -Name $s.Name -ErrorAction SilentlyContinue
            if (-not $svc) { return $null }
            return [string]$svc.StartType
        }
        'mmagent' {
            if ($Sandbox) { return $script:SandboxMC }
            try { return ([string](Get-MMAgent -ErrorAction Stop).MemoryCompression).ToLower() } catch { return $null }
        }
    }
}

function Write-Value($s, $value) {
    switch ($s.Kind) {
        'reg' {
            $t = Resolve-RegTarget $s
            if ($null -eq $value) {
                $k = $t.Root.OpenSubKey($t.Sub, $true)
                if ($k) { try { $k.DeleteValue($s.Name, $false) } finally { $k.Close() } }
                return
            }
            $k = $t.Root.CreateSubKey($t.Sub)
            try {
                if ($s.Type -eq 'DWord') { $k.SetValue($s.Name, [int]$value, [Microsoft.Win32.RegistryValueKind]::DWord) }
                else { $k.SetValue($s.Name, [string]$value, [Microsoft.Win32.RegistryValueKind]::String) }
            } finally { $k.Close() }
        }
        'xml' {
            $doc = New-Object Xml.XmlDocument
            $doc.PreserveWhitespace = $true
            $doc.Load($GtaXml)
            $root = $doc.DocumentElement
            $parent = $root
            if ($s.Section -ne '') {
                $parent = $root.SelectSingleNode($s.Section)
                if (-not $parent) { $parent = $doc.CreateElement($s.Section); [void]$root.AppendChild($parent) }
            }
            $node = $parent.SelectSingleNode($s.Element)
            if ($null -eq $value) { if ($node) { [void]$parent.RemoveChild($node) } }
            else {
                if (-not $node) { $node = $doc.CreateElement($s.Element); [void]$parent.AppendChild($node) }
                if ($s.Element -eq 'configSource') { $node.InnerText = $value } else { $node.SetAttribute('value', $value) }
            }
            $settings = New-Object Xml.XmlWriterSettings
            $settings.Encoding = New-Object Text.UTF8Encoding($false)
            $tmp = $GtaXml + '.aw.tmp'
            $w = [Xml.XmlWriter]::Create($tmp, $settings)
            try { $doc.Save($w) } finally { $w.Close() }
            Move-Item -LiteralPath $tmp -Destination $GtaXml -Force
        }
        'mouse' {
            if ($null -eq $value) { return }
            if ($Sandbox) { $script:SandboxMouse = $value; return }
            $p = $value.Split(',')
            if (-not [NexusNative]::SetMouse([int]$p[0], [int]$p[1], [int]$p[2])) { throw 'SystemParametersInfo failed' }
        }
        'display' {
            if ($null -eq $value) { return }
            if ($Sandbox) { $script:SandboxHz = $value; return }
            $r = [NexusNative]::SetRefresh([int]$value)
            if ($r -ne 0) { throw "Display rejected $value Hz (code $r)" }
        }
        'power' {
            if ($null -eq $value) { return }
            if ($value -eq 'ULTIMATE') {
                # Windows hides Ultimate Performance on most editions; duplicating its template makes it available.
                if ($Sandbox) { $value = 'e9a42b02-d5df-448d-aa00-03f14749eb61' }
                else {
                    $o = (powercfg -duplicatescheme e9a42b02-d5df-448d-aa00-03f14749eb61) | Out-String
                    if ($o -notmatch '([0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12})') { throw 'This Windows edition cannot create Ultimate Performance' }
                    $value = $Matches[1].ToLower()
                }
                $s.Proposed = $value
            }
            if ($Sandbox) { $script:SandboxPower = $value; return }
            powercfg /setactive $value | Out-Null
            if ($LASTEXITCODE -ne 0) { throw 'powercfg /setactive failed' }
        }
        'svc' {
            if ($null -eq $value) { return }
            if ($Sandbox) { $script:SandboxSvc = $value; return }
            Set-Service -Name $s.Name -StartupType $value
            if ($value -ne 'Disabled') { Start-Service -Name $s.Name -ErrorAction SilentlyContinue }
        }
        'mmagent' {
            if ($null -eq $value) { return }
            if ($Sandbox) { $script:SandboxMC = $value; return }
            if ($value -eq 'true') {
                $svc = Get-Service SysMain
                if ([string]$svc.StartType -eq 'Disabled') { throw 'SysMain is still disabled' }
                if ($svc.Status -ne 'Running') { Start-Service SysMain }
                Enable-MMAgent -MemoryCompression
            } else { Disable-MMAgent -MemoryCompression }
        }
    }
}

function Describe($s, $v) {
    if ($null -eq $v) {
        if ($s.Default) { return '(default: ' + (Describe $s $s.Default) + ')' }
        return '(not set)'
    }
    switch ($s.Kind) {
        'reg' {
            if ($s.Id -eq 'hags') { if ($v -eq '2') { return 'On' } return 'Off' }
            if ($s.Type -eq 'DWord') { if ($v -eq '1') { return 'On' } if ($v -eq '0') { return 'Off' } }
            if ($v -like '*GpuPreference=2*') { return 'Dedicated GPU' }
            if ($s.Id -like 'fso:*') { if ($v -match 'DISABLEDXMAXIMIZEDWINDOWEDMODE') { return 'Off' } return "On ($v)" }
            return $v
        }
        'xml' {
            if ($GtaLabels.ContainsKey($s.Element) -and $GtaLabels[$s.Element].ContainsKey([string]$v)) { return $GtaLabels[$s.Element][[string]$v] }
            $d = 0.0
            if ($s.Element -ne 'RefreshRate' -and [double]::TryParse($v, [Globalization.NumberStyles]::Float, $Inv, [ref]$d) -and $v.Contains('.')) { return ('{0:0}%' -f ($d * 100)) }
            return $v
        }
        'mouse' { if ($v.EndsWith(',0')) { return 'Off (1:1)' } return 'On (acceleration)' }
        'svc' { if ($v -eq 'Disabled') { return 'Disabled' } return "Enabled ($v)" }
        'mmagent' { if ($v -eq 'true') { return 'On' } return 'Off' }
        'display' { return "$v Hz" }
        'power' {
            if ($v -eq 'ULTIMATE') { return 'Ultimate Performance (new)' }
            if ($v -eq 'e9a42b02-d5df-448d-aa00-03f14749eb61') { return 'Ultimate Performance' }
            $name = ((powercfg /list) | Out-String) -split "`n" | Where-Object { $_ -match $v } | ForEach-Object { if ($_ -match '\((.+?)\)') { $Matches[1] } } | Select-Object -First 1
            if ($name) { return $name } return $v
        }
    }
    return [string]$v
}

function Test-Support($s, $hw) {
    if ($s.Admin -and -not $IsAdmin -and -not $Sandbox) { return 'needs PowerShell "Run as administrator"' }
    if ($s.Kind -eq 'xml') {
        if (-not (Test-Path $GtaXml)) { return 'gta5_settings.xml not found (start FiveM once first)' }
        if (Test-FiveMRunning) { return 'close FiveM first (it rewrites the file on exit)' }
        if ((Get-Item $GtaXml).IsReadOnly) { return 'gta5_settings.xml is read-only' }
    }
    if ($s.Kind -eq 'power' -and $hw.Laptop -and -not $hw.OnAc) { return 'on battery, plug in the charger first' }
    if ($s.Kind -eq 'svc' -and -not $Sandbox -and -not (Get-Service -Name $s.Name -ErrorAction SilentlyContinue)) { return 'service not present on this Windows edition' }
    if ($s.Kind -eq 'mmagent' -and -not $Sandbox -and -not (Get-Command Enable-MMAgent -ErrorAction SilentlyContinue)) { return 'not supported on this Windows edition' }
    if ($s.Kind -eq 'power' -and $null -eq $s.Proposed) { return 'cannot read the power plan' }
    return $null
}

# ------------------------------------------------------------------ preview / apply / restore
function Build-Preview($plan, $hw) {
    $rows = @()
    foreach ($s in $plan) {
        $cur = $null
        try { $cur = Read-Value $s } catch { $rows += [pscustomobject]@{ S = $s; Cur = $null; Status = 'ERROR'; Note = $_.Exception.Message }; continue }
        if ($s.Kind -eq 'power') {
            # Only stock Windows plans are replaced; custom/OEM performance plans are kept as they are.
            $stock = ($cur -eq $PlanBalanced -or $cur -eq $PlanSaver -or ($s.Ultimate -and $cur -eq $PlanHigh))
            if ($stock) {
                $list = ''; if (-not $Sandbox) { $list = (powercfg /list) | Out-String }
                if ($s.Ultimate) {
                    $existing = $null
                    foreach ($ln in ($list -split "`n")) { if ($ln -match '([0-9a-fA-F-]{36}).*\(Ultimate Performance\)') { $existing = $Matches[1].ToLower(); break } }
                    if ($existing) { $s.Proposed = $existing } else { $s.Proposed = 'ULTIMATE' }
                } else {
                    $hasHigh = $list -match $PlanHigh
                    if ($Sandbox) { $hasHigh = $true }
                    if ($hasHigh) { $s.Proposed = $PlanHigh } else { $s.Proposed = $cur }
                }
            } else { $s.Proposed = $cur }
        }
        if ($s.Id -like 'fso:*') {
            $flag = 'DISABLEDXMAXIMIZEDWINDOWEDMODE'
            if ($null -eq $cur) { $s.Proposed = '~ ' + $flag } elseif ($cur -match $flag) { $s.Proposed = $cur } else { $s.Proposed = $cur.TrimEnd() + ' ' + $flag }
        }
        $cmpCur = $cur; if ($null -eq $cmpCur) { $cmpCur = $s.Default }
        if (Same $cmpCur $s.Proposed) { $rows += [pscustomobject]@{ S = $s; Cur = $cur; Status = 'OK'; Note = '' }; continue }
        $why = Test-Support $s $hw
        if ($why) { $rows += [pscustomobject]@{ S = $s; Cur = $cur; Status = 'SKIP'; Note = $why }; continue }
        $rows += [pscustomobject]@{ S = $s; Cur = $cur; Status = 'CHANGE'; Note = $s.Why }
    }
    return $rows
}

function Show-Preview($rows) {
    $change = @($rows | Where-Object { $_.Status -eq 'CHANGE' })
    $ok = @($rows | Where-Object { $_.Status -eq 'OK' })
    $skip = @($rows | Where-Object { $_.Status -ne 'CHANGE' -and $_.Status -ne 'OK' })
    Head ("{0} to change  |  {1} already optimal  |  {2} skipped" -f $change.Count, $ok.Count, $skip.Count)
    foreach ($r in $change) {
        Say ("  [CHANGE] {0}: {1}  ->  {2}" -f $r.S.Title, (Describe $r.S $r.Cur), (Describe $r.S $r.S.Proposed)) 'Green'
        if ($r.Note) { Say ("           {0}" -f $r.Note) 'DarkGray' }
    }
    foreach ($r in $skip) { Say ("  [SKIP]   {0}: {1}" -f $r.S.Title, $r.Note) 'Yellow' }
    if ($ok.Count -gt 0) { Say ("  [OK]     " + (($ok | ForEach-Object { $_.S.Title -replace '^GTA: ', '' }) -join ', ')) 'DarkGray' }
}

function Invoke-Apply($rows, [string]$label) {
    $todo = @($rows | Where-Object { $_.Status -eq 'CHANGE' })
    if ($todo.Count -eq 0) { Say 'Nothing to change.' 'Green'; return $null }
    New-Item -ItemType Directory -Force -Path $BackupDir | Out-Null
    $stamp = (Get-Date).ToString('yyyyMMdd-HHmmss', $Inv)
    $file = Join-Path $BackupDir "backup-$stamp.json"
    if ($todo | Where-Object { $_.S.Kind -eq 'xml' }) { Copy-Item $GtaXml (Join-Path $BackupDir "gta5_settings-$stamp.xml") }
    $entries = New-Object System.Collections.ArrayList
    $okCount = 0; $fail = 0
    foreach ($r in $todo) {
        $s = $r.S
        $before = Read-Value $s
        $entry = [pscustomobject]@{ Id = $s.Id; Title = $s.Title; Kind = $s.Kind; Hive = $s.Hive; SubKey = $s.SubKey; Name = $s.Name; Type = $s.Type
            Element = $s.Element; Section = $s.Section; Before = $before; After = $s.Proposed; Applied = $false; Verified = $false; Error = $null }
        [void]$entries.Add($entry)
        ConvertTo-Json -InputObject @($entries) -Depth 4 | Set-Content -Path $file -Encoding UTF8   # old value saved before writing
        try {
            Write-Value $s $s.Proposed
            $entry.Applied = $true
            $rb = Read-Value $s
            $entry.After = $s.Proposed   # Write-Value may resolve a placeholder (e.g. a newly created power plan id)
            ConvertTo-Json -InputObject @($entries) -Depth 4 | Set-Content -Path $file -Encoding UTF8
            $entry.Verified = (Same $rb $s.Proposed)
            if ($entry.Verified) { $okCount++; Say ("  OK    {0}: {1}" -f $s.Title, (Describe $s $s.Proposed)) 'Green' }
            else { $fail++; $entry.Error = "read-back = $rb"; Say ("  FAIL  {0}: read back {1}" -f $s.Title, $rb) 'Red' }
        } catch {
            $fail++; $entry.Error = $_.Exception.Message
            Say ("  FAIL  {0}: {1}" -f $s.Title, $_.Exception.Message) 'Red'
        }
        ConvertTo-Json -InputObject @($entries) -Depth 4 | Set-Content -Path $file -Encoding UTF8
    }
    Head ("Done: {0} applied, {1} failed" -f $okCount, $fail)
    Say ("  Backup: $file") 'Cyan'
    if ($todo | Where-Object { $_.S.Kind -eq 'xml' }) { Say '  Game graphics apply the next time FiveM starts.' 'Gray' }
    if ($todo | Where-Object { $_.S.Id -eq 'hags' }) { Say '  HAGS takes effect after a reboot.' 'Yellow' }
    if ($todo | Where-Object { $_.S.Id -in @('memcomp', 'sysmain') }) { Say '  Memory Compression is fully active after one reboot.' 'Yellow' }
    return $file
}

function Invoke-Restore([string]$file) {
    $parsed = ConvertFrom-Json -InputObject (Get-Content $file -Raw -Encoding UTF8)  # not via pipeline: PS 5.1 would nest the array
    $entries = @($parsed)
    $ok = 0; $fail = 0
    [array]::Reverse($entries)
    foreach ($e in $entries) {
        if (-not $e.Applied) { continue }
        $s = [pscustomobject]@{ Id = $e.Id; Title = $e.Title; Kind = $e.Kind; Hive = $e.Hive; SubKey = $e.SubKey; Name = $e.Name; Type = $e.Type
            Element = $e.Element; Section = $e.Section; Default = $null; Admin = ($e.Hive -eq 'HKLM' -or $e.Kind -in @('svc', 'mmagent')) }
        if ($s.Kind -eq 'xml' -and (Test-FiveMRunning)) { $fail++; Say ("  SKIP  {0}: close FiveM first" -f $s.Title) 'Yellow'; continue }
        if ($s.Admin -and -not $IsAdmin -and -not $Sandbox) { $fail++; Say ("  SKIP  {0}: needs administrator" -f $s.Title) 'Yellow'; continue }
        try {
            Write-Value $s $e.Before
            if (Same (Read-Value $s) $e.Before) { $ok++; Say ("  RESTORED {0}: {1}" -f $s.Title, (Describe $s $e.Before)) 'Green' }
            else { $fail++; Say ("  FAIL  {0}: read-back mismatch" -f $s.Title) 'Red' }
        } catch { $fail++; Say ("  FAIL  {0}: {1}" -f $s.Title, $_.Exception.Message) 'Red' }
    }
    if ($fail -eq 0) { Rename-Item -LiteralPath $file -NewName ([IO.Path]::GetFileNameWithoutExtension($file) + '.restored.json') }
    Head ("Restore: {0} restored, {1} failed" -f $ok, $fail)
}

function Show-Scan($hw) {
    Head 'PC specs'
    Say ("  CPU      : {0} ({1} threads)" -f $hw.Cpu, $hw.Threads)
    foreach ($g in $hw.Gpus) {
        $tag = ''; if (-not $g.IsPci) { $tag = ' [virtual]' } elseif ($g.Integrated) { $tag = ' [integrated]' }
        Say ("  GPU      : {0} {1} GB{2}" -f $g.Name, $g.VramGb, $tag)
    }
    Say ("  RAM      : {0} GB" -f $hw.RamGb)
    Say ("  Display  : {0}x{1} @ {2} Hz (max {3} Hz)" -f $hw.Width, $hw.Height, $hw.Hz, $hw.MaxHz)
    Say ("  Windows  : {0}" -f $hw.Windows)
    $ssdText = 'unknown'; if ($hw.Ssd -eq $true) { $ssdText = 'SSD' } elseif ($hw.Ssd -eq $false) { $ssdText = 'HDD' }
    Say ("  Disk     : {0}   |  PC tier: {1}" -f $ssdText, $hw.Tier) 'White'
    $warn = @()
    if ($hw.RamGb -lt 15) { $warn += "Only $($hw.RamGb) GB RAM: close Chrome, Discord overlay and other apps before playing" }
    elseif ($hw.RamGb -lt 24) { $warn += '16 GB RAM works, but servers with many custom assets get tight: close browsers while playing' }
    if ($hw.MaxHz -gt $hw.Hz) { $warn += "Monitor supports $($hw.MaxHz) Hz but is set to $($hw.Hz) Hz" }
    if ($hw.Hybrid) { $warn += 'Integrated GPU detected: FiveM must be pinned to the dedicated GPU (included in every profile)' }
    if ($hw.Ssd -eq $false) { $warn += 'FiveM is on an HDD: expect stutter while assets load, an SSD is strongly recommended' }
    if ($hw.Laptop -and -not $hw.OnAc) { $warn += 'Laptop on battery: plug in the charger before playing' }
    $os = Get-CimInstance Win32_OperatingSystem
    $free = [math]::Round($os.FreePhysicalMemory / 1MB, 1)
    if ($free -lt 2) { $warn += "Only $free GB RAM free right now" }
    if ($warn.Count -gt 0) { Head 'Warnings'; foreach ($w in $warn) { Say "  - $w" 'Yellow' } }
    $rec = 'MEDIUM'; if ($hw.Tier -eq 'LOW') { $rec = 'LOW' } elseif ($hw.Tier -eq 'HIGH') { $rec = 'MAX' }
    Say ''
    Say "  Recommended level for this PC: $rec" 'Green'
    if (Test-FiveMRunning) { Say '  FiveM is running: game graphics can be changed after closing it.' 'Yellow' }
}

function Show-Apps {
    Head 'Apps using the most RAM / CPU (2 s sample)'
    $a = @{}; Get-Process | ForEach-Object { try { $a[$_.Id] = $_.TotalProcessorTime.TotalMilliseconds } catch { } }
    Start-Sleep -Seconds 2
    $cores = [Environment]::ProcessorCount
    $groups = @{}
    foreach ($p in Get-Process) {
        $cpu = 0
        try { if ($a.ContainsKey($p.Id)) { $cpu = ($p.TotalProcessorTime.TotalMilliseconds - $a[$p.Id]) / 2000 / $cores * 100 } } catch { }
        if (-not $groups.ContainsKey($p.Name)) { $groups[$p.Name] = [pscustomobject]@{ Name = $p.Name; Count = 0; Cpu = 0.0; Mem = 0.0 } }
        $g = $groups[$p.Name]; $g.Count++; $g.Cpu += [math]::Max(0, $cpu); $g.Mem += $p.WorkingSet64 / 1MB
    }
    $tips = @{ 'chrome' = 'browser: close while playing'; 'msedge' = 'browser: close while playing'; 'firefox' = 'browser: close while playing'; 'opera' = 'browser: close while playing'
        'Discord' = 'disable Discord overlay and hardware acceleration'; 'Medal' = 'background clip recording: close if unused'; 'MedalEncoder' = 'background clip recording: close if unused'
        'obs64' = 'uses the GPU encoder: close if not streaming'; 'EpicGamesLauncher' = 'not needed by FiveM'; 'Overwolf' = 'overlay can cause stutter'
        'iCUE' = 'RGB software can cause micro-stutter'; 'LightingService' = 'RGB software can cause micro-stutter'; 'wallpaper64' = 'animated wallpaper uses the GPU'; 'OneDrive' = 'pause sync while playing' }
    $groups.Values | Where-Object { $_.Name -notin @('Idle', 'System', 'Memory Compression') } | Sort-Object { $_.Mem + $_.Cpu * 50 } -Descending | Select-Object -First 12 | ForEach-Object {
        $tip = ''; if ($tips.ContainsKey($_.Name)) { $tip = '  <- ' + $tips[$_.Name] }
        $color = 'Gray'; if ($tip) { $color = 'Yellow' }
        Say ("  {0,-28} {1,6:0.0}% CPU {2,7:0} MB{3}" -f ($_.Name + ' x' + $_.Count), $_.Cpu, $_.Mem, $tip) $color
    }
    Say '  (This view closes nothing. Menu 8 / 9 can close RAM-heavy apps after asking.)' 'DarkGray'
}

function Invoke-Monitor([int]$seconds = 30) {
    Head "Measuring for $seconds s (keep playing meanwhile)"
    Say '  Note: FPS needs PresentMon and is not shown here. In game use F8 > cl_drawfps 1' 'DarkGray'
    $smi = Get-Command nvidia-smi -ErrorAction SilentlyContinue
    $hwMon = $hw
    $cpuS = @(); $gpuS = @(); $vramS = @(); $tempS = @(); $freeS = @(); $gameS = @()
    $cores = [Environment]::ProcessorCount
    $lastGame = $null; $lastT = $null
    for ($i = 1; $i -le $seconds; $i++) {
        try { $cpuS += (Get-CimInstance Win32_PerfFormattedData_PerfOS_Processor -Filter "Name='_Total'").PercentProcessorTime } catch { }
        try { $freeS += (Get-CimInstance Win32_OperatingSystem).FreePhysicalMemory / 1MB } catch { }
        $gotGpu = $false
        if ($smi) {
            try {
                $line = (& nvidia-smi --query-gpu=utilization.gpu,memory.used,memory.total,temperature.gpu --format=csv,noheader,nounits | Select-Object -First 1)
                $p = $line -split ',\s*'
                if ($p.Count -ge 4) { $gpuS += [double]$p[0]; $vramS += [double]$p[1] / [double]$p[2] * 100; $tempS += [double]$p[3]; $gotGpu = $true }
            } catch { }
        }
        if (-not $gotGpu) {
            # AMD / Intel / any GPU: Windows GPU counters (Windows 10 1709+). Temperature is not available this way.
            try {
                $eng = Get-CimInstance Win32_PerfFormattedData_GPUPerformanceCounters_GPUEngine -ErrorAction Stop | Where-Object { $_.Name -like '*engtype_3D*' }
                if ($eng) {
                    $byAdapter = @{}
                    foreach ($e in $eng) { $luid = ($e.Name -replace '^.*?(luid_[^_]+_[^_]+).*$', '$1'); $byAdapter[$luid] = [double]$byAdapter[$luid] + [double]$e.UtilizationPercentage }
                    $gpuS += [math]::Min(100, ($byAdapter.Values | Measure-Object -Maximum).Maximum)
                }
                if ($hwMon.VramGb -gt 0) {
                    $mem = Get-CimInstance Win32_PerfFormattedData_GPUPerformanceCounters_GPUAdapterMemory -ErrorAction Stop | Measure-Object DedicatedUsage -Maximum
                    if ($mem.Maximum) { $vramS += [math]::Min(100, $mem.Maximum / 1GB / $hwMon.VramGb * 100) }
                }
            } catch { }
        }
        $g = Get-Process | Where-Object { $_.Name -like '*_GTAProcess' } | Select-Object -First 1
        if ($g) {
            try {
                $now = $g.TotalProcessorTime.TotalMilliseconds; $t = Get-Date
                if ($null -ne $lastGame) { $gameS += ($now - $lastGame) / ($t - $lastT).TotalMilliseconds / $cores * 100 }
                $lastGame = $now; $lastT = $t
            } catch { }
        }
        Write-Progress -Activity "$Brand monitor" -Status "$i / $seconds s" -PercentComplete ($i / $seconds * 100)
        Start-Sleep -Milliseconds 700
    }
    Write-Progress -Activity "$Brand monitor" -Completed
    function St($a, $fmt) { if ($a.Count -eq 0) { return 'N/A' } $m = $a | Measure-Object -Average -Maximum; return ("avg {0}  max {1}" -f ($m.Average.ToString($fmt)), ($m.Maximum.ToString($fmt))) }
    Say ("  CPU total       : {0} %" -f (St $cpuS '0'))
    Say ("  FiveM CPU       : {0} %" -f (St $gameS '0.0'))
    Say ("  GPU             : {0} %" -f (St $gpuS '0'))
    Say ("  VRAM used       : {0} %" -f (St $vramS '0'))
    Say ("  GPU temperature : {0} C" -f (St $tempS '0'))
    $minFree = 'N/A'; if ($freeS.Count) { $minFree = ($freeS | Measure-Object -Minimum).Minimum.ToString('0.0') }
    Say ("  Lowest free RAM : {0} GB" -f $minFree)
    Head 'Analysis'
    $found = $false
    $gpuAvg = 0; if ($gpuS.Count) { $gpuAvg = ($gpuS | Measure-Object -Average).Average }
    if ($gpuS.Count -and $gpuAvg -ge 95) { Say '  GPU-bound: lower Shadow / Grass / MSAA / Post FX.' 'Yellow'; $found = $true }
    if ($vramS.Count -and ($vramS | Measure-Object -Average).Average -ge 95) { Say '  VRAM full: lower Texture Quality by one step.' 'Yellow'; $found = $true }
    if ($freeS.Count -and ($freeS | Measure-Object -Minimum).Minimum -lt 1.5) { Say '  RAM too low: Windows pages to disk, which causes stutter. Close other apps (menu 7).' 'Yellow'; $found = $true }
    if ($gpuS.Count -and $gpuAvg -lt 80 -and $gameS.Count) { Say '  GPU not saturated: the limit is likely CPU / server scripts. Lower distance scaling / population.' 'Yellow'; $found = $true }
    if ($gameS.Count -eq 0) { Say '  FiveM was not running: these numbers are for the desktop, not the game.' 'Yellow'; $found = $true }
    if (-not $found) { Say '  No clear bottleneck.' 'Green' }
}

function Invoke-CacheClean {
    if (-not (Test-Path $FiveMData)) { Say 'FiveM folder not found.' 'Yellow'; return }
    if (Test-FiveMRunning) { Say 'Close FiveM completely first (including the launcher).' 'Yellow'; return }
    Head 'Clean FiveM cache'
    Say '  Cleaning cache does not raise FPS. It fixes missing textures and crashes when joining.' 'Gray'
    Say '  Server files are downloaded again on the next join. game-storage and nui-storage are never deleted.' 'Gray'
    $targets = @()
    $cache = Join-Path $FiveMData 'cache'
    if (Test-Path $cache) { $targets += @(Get-ChildItem $cache -Force | Where-Object { $_.Name -notin @('subprocess', 'game') }) }
    foreach ($n in 'server-cache', 'server-cache-priv') { $p = Join-Path $FiveMData $n; if (Test-Path $p) { $targets += @(Get-ChildItem $p -Force) } }
    $size = 0; foreach ($t in $targets) { $size += (Get-ChildItem $t.FullName -Recurse -Force -File -ErrorAction SilentlyContinue | Measure-Object Length -Sum).Sum; if (-not $t.PSIsContainer) { $size += $t.Length } }
    Say ("  Size to delete: {0:0.0} GB" -f ($size / 1GB)) 'White'
    $ans = Read-Host '  Type YES to delete (cannot be undone)'
    if ($ans -cne 'YES') { Say '  Cancelled.' 'Gray'; return }
    $err = 0
    foreach ($t in $targets) { try { Remove-Item -LiteralPath $t.FullName -Recurse -Force -Confirm:$false -ErrorAction Stop } catch { $err++ } }
    Say ("  Done ({0} items could not be deleted)" -f $err) 'Green'
}

# Apps that are safe to close for a gaming session (never system processes, never Discord/Steam/FiveM).
$CloseableApps = [ordered]@{
    'chrome' = 'Chrome'; 'msedge' = 'Edge'; 'firefox' = 'Firefox'; 'opera' = 'Opera'; 'brave' = 'Brave'
    'Medal' = 'Medal'; 'MedalEncoder' = 'Medal encoder'; 'Overwolf' = 'Overwolf'; 'EpicGamesLauncher' = 'Epic Games'
    'EADesktop' = 'EA app'; 'Battle.net' = 'Battle.net'; 'wallpaper64' = 'Wallpaper Engine'; 'wallpaper32' = 'Wallpaper Engine'
    'Spotify' = 'Spotify'; 'OneDrive' = 'OneDrive'
}

function Get-RamApps([string[]]$Keep = @()) {
    $found = @()
    if ($Keep -contains '*') { return $found }
    foreach ($name in $CloseableApps.Keys) {
        if ($Keep -contains $name) { continue }
        $p = @(Get-Process -Name $name -ErrorAction SilentlyContinue)
        if ($p.Count -gt 0) { $found += [pscustomobject]@{ Name = $name; Label = $CloseableApps[$name]; Count = $p.Count; Mb = [math]::Round((($p | Measure-Object WorkingSet64 -Sum).Sum) / 1MB) } }
    }
    return $found
}

function Show-RamApps($found) {
    if ($found.Count -eq 0) { Say '  No RAM-heavy apps to close.' 'Green'; return }
    $total = ($found | Measure-Object Mb -Sum).Sum
    Head ("RAM-heavy apps ({0} MB total)" -f $total)
    foreach ($a in $found) { Say ("  - {0}  {1} MB" -f $a.Label, $a.Mb) 'Yellow' }
    Say '  Save any unsaved work in these apps first (Chrome restores its tabs when reopened).' 'DarkGray'
}

function Close-BackgroundApps([switch]$Confirmed, [string[]]$Keep = @()) {
    $found = @(Get-RamApps -Keep $Keep)
    if (-not $Confirmed) {
        Show-RamApps $found
        if ($found.Count -eq 0 -or $AutoAction) { return }
        $yes = RH '  Close these apps now? (y/n)'
        if ($yes -ne 'y') { Say '  Nothing closed.' 'Gray'; return }
    }
    if ($found.Count -eq 0) { return }
    if ($Sandbox) { Say ('  SANDBOX: simulated closing ' + (($found | ForEach-Object { $_.Label }) -join ', ')) 'Magenta'; return }
    $before = (Get-CimInstance Win32_OperatingSystem).FreePhysicalMemory / 1MB
    foreach ($a in $found) { Get-Process -Name $a.Name -ErrorAction SilentlyContinue | ForEach-Object { try { $_.CloseMainWindow() | Out-Null } catch { } } }
    Start-Sleep -Seconds 3
    foreach ($a in $found) { Get-Process -Name $a.Name -ErrorAction SilentlyContinue | Stop-Process -Force -ErrorAction SilentlyContinue }
    Start-Sleep -Seconds 2
    $after = (Get-CimInstance Win32_OperatingSystem).FreePhysicalMemory / 1MB
    Say ("  Closed. Free RAM: {0:0.0} GB -> {1:0.0} GB" -f $before, $after) 'Green'
}

# Checks presets and plans against made-up PCs (read-only): every tier, GPU vendor and form factor.
function Invoke-SelfTest {
    $fake = @(
        @{ N = 'Low-end laptop, Intel UHD, 8GB, battery'; Tier = 'LOW'; VramGb = 0; Ram = 8; Vendor = 'Intel'; Hybrid = $false; Laptop = $true; OnAc = $false; Hz = 60; MaxHz = 60; Build = 19045 }
        @{ N = 'Gaming laptop, RTX 3050 4GB + iGPU, 16GB'; Tier = 'MID'; VramGb = 4; Ram = 16; Vendor = 'NVIDIA'; Hybrid = $true; Laptop = $true; OnAc = $true; Hz = 60; MaxHz = 144; Build = 22631 }
        @{ N = 'Desktop, RX 6600 8GB, 16GB, Win11'; Tier = 'HIGH'; VramGb = 8; Ram = 16; Vendor = 'AMD'; Hybrid = $false; Laptop = $false; OnAc = $true; Hz = 165; MaxHz = 165; Build = 26100 }
        @{ N = 'Desktop, RTX 4090 24GB, 64GB'; Tier = 'HIGH'; VramGb = 24; Ram = 64; Vendor = 'NVIDIA'; Hybrid = $false; Laptop = $false; OnAc = $true; Hz = 240; MaxHz = 240; Build = 19045 }
        @{ N = 'Old Win10 1909, GTX 1050 2GB, 8GB'; Tier = 'LOW'; VramGb = 2; Ram = 8; Vendor = 'NVIDIA'; Hybrid = $false; Laptop = $false; OnAc = $true; Hz = 75; MaxHz = 75; Build = 18363 }
    )
    $fail = 0
    foreach ($f in $fake) {
        $gpu = [pscustomobject]@{ Name = 'fake'; Vendor = $f.Vendor; VramGb = $f.VramGb; IsPci = $true; Integrated = ($f.VramGb -eq 0); Driver = '' }
        $hwF = [pscustomobject]@{ Cpu = 'fake'; Threads = 8; RamGb = $f.Ram; Gpus = @($gpu); Primary = $gpu; VramGb = $f.VramGb; Hybrid = $f.Hybrid
            Laptop = $f.Laptop; OnAc = $f.OnAc; Width = 1920; Height = 1080; Hz = $f.Hz; MaxHz = $f.MaxHz; Windows = 'fake'; Build = $f.Build; Tier = $f.Tier; Ssd = $true }
        $lvl = Get-BestSelection $hwF
        foreach ($combo in @(@('LOW', 'MAXFPS'), @('MEDIUM', 'BALANCED'), @('MAX', 'CROWD'))) {
            try {
                $plan = @(Get-Plan $combo[0] $combo[1] $hwF)
                $preset = Get-GtaPreset $combo[1] $hwF
                $tex = $preset['TextureQuality']
                $hags = [bool]($plan | Where-Object { $_.Id -eq 'hags' })
                $refresh = [bool]($plan | Where-Object { $_.Id -eq 'refresh' })
                $ok = $true
                if ($f.VramGb -lt 5.5 -and $tex -ne '0' -and $combo[1] -ne 'BALANCED') { $ok = $false }
                if ($hags -and ($f.Build -lt 19041 -or $f.Vendor -eq 'Intel')) { $ok = $false }
                if ($refresh -ne ($f.MaxHz -gt $f.Hz)) { $ok = $false }
                if ($f.Hz -gt 0 -and $preset['RefreshRate'] -ne [string]$f.MaxHz) { $ok = $false }
                if (-not $ok) { $fail++ }
                $mark = 'OK  '; $col = 'Green'; if (-not $ok) { $mark = 'FAIL'; $col = 'Red' }
                Say ("  {0} {1,-44} {2,-6}/{3,-8} items={4,2} tex={5} hags={6} refresh={7}" -f $mark, $f.N, $combo[0], $combo[1], $plan.Count, $tex, $hags, $refresh) $col
            } catch { $fail++; Say ("  FAIL {0} {1}: {2}" -f $f.N, $combo[0], $_.Exception.Message) 'Red' }
        }
        Say ("       best for this PC: {0} + {1}" -f $lvl[0], $lvl[1]) 'DarkGray'
    }
    Say ("SELFTEST: {0} failures" -f $fail) 'Cyan'
}

function Select-Backup {

    }) | Sort-Object Name -Descending
    $files = @($files)
    if ($files.Count -eq 0) { Say 'No backups left to restore.' 'Yellow'; return $null }

# Preview: graphics rows that must wait for FiveM are folded into one line that lists this Setting's own values
function Show-Preview($rows) {
    $change = @($rows | Where-Object { $_.Status -eq 'CHANGE' })
    $ok = @($rows | Where-Object { $_.Status -eq 'OK' })
    $skip = @($rows | Where-Object { $_.Status -ne 'CHANGE' -and $_.Status -ne 'OK' })
    $win = @($rows | Where-Object { $_.S.Kind -ne 'xml' }).Count; $gx = @($rows | Where-Object { $_.S.Kind -eq 'xml' }).Count
    Head ("{0} to change  |  {1} already optimal  |  {2} skipped" -f $change.Count, $ok.Count, $skip.Count)
    Say ("  {0} adjusts {1} items on this PC: {2} Windows + {3} game graphics" -f $Theme.Name, $rows.Count, $win, $gx) 'White'
    foreach ($r in $change) {
        Say ("  [CHANGE] {0}: {1}  ->  {2}" -f $r.S.Title, (Describe $r.S $r.Cur), (Describe $r.S $r.S.Proposed)) 'Green'
        if ($r.Note) { Say ("           {0}" -f $r.Note) 'DarkGray' }
    }
    $gfx = @($skip | Where-Object { $_.S.Kind -eq 'xml' })
    foreach ($r in @($skip | Where-Object { $_.S.Kind -ne 'xml' })) { Say ("  [SKIP]   {0}: {1}" -f $r.S.Title, $r.Note) 'Yellow' }
    if ($gfx.Count -gt 0) {
        Say ("  [SKIP]   {0} game graphics values wait: {1}" -f $gfx.Count, $gfx[0].Note) 'Yellow'
        $vals = @($gfx | Where-Object { $_.S.Element -ne 'configSource' } | ForEach-Object { $_.S.Element + ' ' + (Describe $_.S $_.S.Proposed) })
        Say ('           ' + $Theme.Name + ' graphics: ' + ($vals -join ', ')) 'DarkGray'
    }
    if ($ok.Count -gt 0) {
        Say ("  [OK]     {0} items already at the {1} value:" -f $ok.Count, $Theme.Name) 'DarkGray'
        foreach ($r in $ok) { Say ("           {0} = {1}" -f ($r.S.Title -replace '^GTA: ', 'Game '), (Describe $r.S $r.S.Proposed)) 'DarkGray' }
        Say '           (some may come from a Setting you ran before: run REWIND first to start clean)' 'DarkGray'
    }
}

${function:Get-PlanBase} = ${function:Get-Plan}
function Get-Plan([string]$profile, [string]$mode, $hw) {
    $plan = @(Get-PlanBase $profile $mode $hw)
    $plan = @($plan | Where-Object { $id = $_.Id; @($Recipe.Drop | Where-Object { $id -eq $_ -or $id -like ($_ + ':*') }).Count -eq 0 })
    if (@($plan | Where-Object { $_.Kind -eq 'xml' }).Count -gt 0) {
        foreach ($k in $Recipe.Gta.Keys) {
            $row = @($plan | Where-Object { $_.Id -eq ('gta.' + $k) }) | Select-Object -First 1
            if ($row) { $row.Proposed = $Recipe.Gta[$k]; continue }
            $section = 'graphics'; if ($VideoElements -contains $k) { $section = 'video' }
            $plan += [pscustomobject]@{ Id = 'gta.' + $k; Title = 'GTA: ' + $k; Kind = 'xml'; Element = $k; Section = $section; Proposed = $Recipe.Gta[$k]; Default = $null; Why = ''; Admin = $false }
        }
    }
    $plan += @(Get-SignatureRows $hw)
    return $plan
}
